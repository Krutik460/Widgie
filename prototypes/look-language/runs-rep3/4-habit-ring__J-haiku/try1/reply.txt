function lerpColor(a, b, t) {
  const pa = [1, 3, 5].map(i => parseInt(a.slice(i, i + 2), 16));
  const pb = [1, 3, 5].map(i => parseInt(b.slice(i, i + 2), 16));
  return '#' + pa.map((v, i) => Math.round(v + (pb[i] - v) * t).toString(16).padStart(2, '0')).join('');
}

function render(ctx) {
  const d = ctx.data;
  const days = d.days || [];
  const doneCount = days.filter(x => x.done).length;
  const progress = Math.min(Math.max(d.streak / 30, 0), 1);
  const ringColor = lerpColor('#E63946', '#2BD36B', progress);
  const ringSize = ctx.family === 'medium' ? 120 : 110;

  const ring = {
    type: 'zstack',
    children: [
      { type: 'circle', stroke: { color: '#FFFFFF22', width: 10, cap: 'round' }, frame: { width: ringSize, height: ringSize } },
      {
        type: 'circle',
        stroke: { color: ringColor, width: 10, cap: 'round' },
        trimFrom: 0,
        trimTo: Math.max(progress, 0.02),
        rotation: -90,
        frame: { width: ringSize, height: ringSize },
      },
      {
        type: 'vstack',
        spacing: 0,
        children: [
          { type: 'symbol', name: 'flame.fill', size: 22, color: ringColor },
          { type: 'text', value: String(d.streak), size: 26, weight: 'heavy', design: 'rounded', color: '#FFFFFF' },
        ],
      },
    ],
  };

  const name = { type: 'text', value: d.name, size: 14, weight: 'semibold', color: '#FFFFFF', lineLimit: 1, minScale: 0.6 };

  const status = {
    type: 'text',
    value: d.doneToday ? 'Done today' : 'Not yet today',
    size: 11,
    weight: 'medium',
    color: d.doneToday ? '#7AE582' : '#9FB3C8',
  };

  const grid = {
    type: 'vstack',
    spacing: 3,
    children: [0, 1, 2, 3, 4].map(row => ({
      type: 'hstack',
      spacing: 3,
      children: [0, 1, 2, 3, 4, 5, 6].map(col => {
        const day = days[row * 7 + col];
        const isToday = row * 7 + col === days.length - 1;
        return {
          type: 'rect',
          cornerRadius: 2,
          fill: day && day.done ? '#7AE582' : '#FFFFFF22',
          stroke: isToday ? { color: '#FFFFFF', width: 1, cap: 'butt' } : undefined,
          frame: { width: 8, height: 8 },
        };
      }),
    })),
  };

  const toggle = {
    type: 'button',
    action: 'toggleToday',
    children: [
      {
        type: 'symbol',
        name: d.doneToday ? 'checkmark' : 'plus',
        size: 14,
        weight: 'bold',
        color: '#0C1821',
        frame: { width: 30, height: 30 },
        background: '#7AE582',
        clip: 'circle',
      },
    ],
  };

  if (ctx.family === 'medium') {
    return {
      type: 'hstack',
      spacing: 16,
      padding: 14,
      frame: { maxWidth: 'infinity', maxHeight: 'infinity' },
      widgetBackground: { gradient: 'linear', stops: ['#1B2A41', '#0C1821'], from: 'top', to: 'bottom' },
      children: [
        ring,
        {
          type: 'vstack',
          align: 'leading',
          spacing: 8,
          children: [
            name,
            status,
            { type: 'text', value: doneCount + ' of 35 days', size: 11, color: '#9FB3C8' },
            { type: 'spacer' },
            grid,
            {
              type: 'hstack',
              children: [{ type: 'spacer' }, toggle],
            },
          ],
        },
      ],
    };
  }

  return {
    type: 'vstack',
    align: 'center',
    spacing: 8,
    padding: 12,
    frame: { maxWidth: 'infinity', maxHeight: 'infinity' },
    widgetBackground: { gradient: 'linear', stops: ['#1B2A41', '#0C1821'], from: 'top', to: 'bottom' },
    children: [
      name,
      ring,
      status,
      {
        type: 'hstack',
        children: [{ type: 'spacer' }, toggle],
      },
    ],
  };
}
