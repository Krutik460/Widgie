function clamp(v, a, b) {
  return Math.min(Math.max(v, a), b);
}

function render(ctx) {
  const d = ctx.data;
  const medium = ctx.family === 'medium';
  const goal = d.goal > 0 ? d.goal : 1;
  const count = Math.max(d.todayCount || 0, 0);
  const frac = clamp(count / goal, 0, 1);
  const done = count >= goal;

  const gw = medium ? 74 : 56;
  const gh = medium ? 122 : 100;
  const inner = gh - 6;
  const wh = count > 0 ? Math.max(8, inner * frac) : 0;

  const waterTop = done ? '#5CE1FF' : '#4CC3FF';
  const waterBot = done ? '#1E88FF' : '#1A6FE0';

  const bubbleDefs = [
    { x: -0.18, y: 0.25, s: 9 },
    { x: 0.2, y: 0.5, s: 6 },
    { x: -0.05, y: 0.75, s: 8 },
    { x: 0.24, y: 0.15, s: 5 },
  ];
  const bubbles = bubbleDefs.map(b => {
    const by = wh * b.y;
    return wh > 24 && {
      type: 'circle',
      stroke: { color: '#FFFFFFCC', width: 1.5 },
      fill: '#FFFFFF33',
      frame: { width: b.s, height: b.s },
      offset: { x: gw * b.x, y: -by - 4 },
    };
  });

  const water = {
    type: 'zstack',
    align: 'bottom',
    frame: { width: gw - 6, height: inner, alignment: 'bottom' },
    children: [
      wh > 0 && {
        type: 'rect',
        fill: { gradient: 'linear', stops: [waterTop, waterBot], from: 'top', to: 'bottom' },
        frame: { width: gw - 6, height: wh },
      },
      wh > 0 && {
        type: 'capsule',
        fill: '#FFFFFF66',
        frame: { width: gw - 18, height: 4 },
        offset: { x: 0, y: -wh + 6 },
      },
      bubbles,
    ],
    cornerRadius: 12,
  };

  const glass = {
    type: 'zstack',
    frame: { width: gw, height: gh },
    children: [
      { type: 'rect', cornerRadius: 15, fill: '#FFFFFF22', frame: { width: gw, height: gh } },
      water,
      {
        type: 'rect',
        cornerRadius: 15,
        stroke: { color: '#FFFFFFAA', width: 3 },
        frame: { width: gw, height: gh },
      },
    ],
  };

  const countRow = {
    type: 'hstack',
    align: 'firstBaseline',
    spacing: 2,
    children: [
      { type: 'text', value: String(count), size: medium ? 46 : 36, weight: 'heavy', design: 'rounded', color: '#FFFFFF', lineLimit: 1, minScale: 0.6 },
      { type: 'text', value: '/' + d.goal, size: medium ? 20 : 16, weight: 'semibold', design: 'rounded', color: '#BFE6FF', lineLimit: 1 },
    ],
  };

  const undo = {
    type: 'button',
    action: 'undoLast',
    children: [
      {
        type: 'symbol', name: 'arrow.uturn.backward', size: 12, weight: 'bold', color: '#FFFFFF',
        frame: { width: 26, height: 26 }, background: '#FFFFFF33', clip: 'circle',
      },
    ],
  };

  const unitRow = {
    type: 'hstack',
    spacing: 4,
    children: [
      done
        ? { type: 'symbol', name: 'checkmark.circle.fill', size: 13, color: '#B6FFE0' }
        : null,
      { type: 'text', value: done ? 'Goal reached' : String(d.unit), size: 12, weight: 'medium', design: 'rounded', color: '#BFE6FF', lineLimit: 1, minScale: 0.7 },
      { type: 'spacer' },
      undo,
    ],
  };

  const last7 = Array.isArray(d.last7) ? d.last7 : [];
  const maxV = Math.max(goal, ...last7, 1);
  const bars = {
    type: 'hstack',
    align: 'bottom',
    spacing: 6,
    frame: { height: 44, alignment: 'bottom' },
    children: last7.map((v, i) => ({
      type: 'capsule',
      fill: i === last7.length - 1 ? '#FFFFFF' : (v >= goal ? '#7FE3FF' : '#FFFFFF55'),
      frame: { width: 12, height: 5 + 39 * clamp(v / maxV, 0, 1) },
    })),
  };

  const plusSize = medium ? 78 : 58;
  const plus = {
    type: 'button',
    action: 'addGlass',
    children: [
      {
        type: 'symbol', name: 'plus', size: medium ? 38 : 28, weight: 'heavy', color: '#0A5BD6',
        frame: { width: plusSize, height: plusSize },
        background: { gradient: 'linear', stops: ['#FFFFFF', '#CFEFFF'], from: 'top', to: 'bottom' },
        clip: 'circle',
        shadow: { color: '#0A3A8A88', radius: 6, x: 0, y: 3 },
      },
    ],
  };

  const info = {
    type: 'vstack',
    align: 'leading',
    spacing: medium ? 6 : 4,
    frame: { maxWidth: 'infinity', maxHeight: 'infinity', alignment: 'leading' },
    children: medium
      ? [countRow, unitRow, { type: 'spacer' }, bars]
      : [countRow, unitRow, { type: 'spacer' }, { type: 'hstack', children: [{ type: 'spacer' }, plus] }],
  };

  const decorBubbles = [
    { x: 0.88, y: 0.12, s: 22 },
    { x: 0.7, y: 0.9, s: 14 },
    { x: 0.95, y: 0.6, s: 10 },
  ].map(b => ({
    type: 'circle',
    fill: '#FFFFFF14',
    frame: { width: b.s, height: b.s },
    offset: { x: (b.x - 0.5) * ctx.width, y: (b.y - 0.5) * ctx.height },
  }));

  const content = {
    type: 'hstack',
    align: 'center',
    spacing: medium ? 16 : 10,
    padding: 12,
    frame: { maxWidth: 'infinity', maxHeight: 'infinity', alignment: 'center' },
    children: medium
      ? [glass, info, { type: 'vstack', frame: { maxHeight: 'infinity', alignment: 'center' }, children: [plus] }]
      : [glass, info],
  };

  return {
    type: 'zstack',
    frame: { maxWidth: 'infinity', maxHeight: 'infinity' },
    widgetBackground: { gradient: 'linear', stops: ['#2AA7FF', '#0B4FC9'], from: 'topLeading', to: 'bottomTrailing' },
    children: [decorBubbles, content],
  };
}
