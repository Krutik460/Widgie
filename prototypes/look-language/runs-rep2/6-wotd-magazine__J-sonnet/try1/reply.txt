function formatDate(s) {
  const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
  const p = String(s || '').split('-');
  if (p.length !== 3) return String(s || '');
  const m = parseInt(p[1], 10);
  const dd = parseInt(p[2], 10);
  if (!(m >= 1 && m <= 12) || isNaN(dd)) return String(s);
  return months[m - 1].toUpperCase() + ' ' + dd;
}

function render(ctx) {
  const d = ctx.data;
  const medium = ctx.family === 'medium';
  const word = String(d.word || '');
  const first = word.charAt(0).toUpperCase();
  const rest = word.slice(1);

  const CREAM = '#F6EEDC';
  const RED = '#7A1420';
  const DEEP = '#4A0A12';

  const header = {
    type: 'hstack',
    spacing: 6,
    children: [
      { type: 'text', value: formatDate(d.date), size: 8, weight: 'semibold', design: 'serif', color: RED, kerning: 1.5, lineLimit: 1 },
      { type: 'spacer' },
      { type: 'text', value: 'WORD OF THE DAY', size: 8, weight: 'semibold', design: 'serif', color: RED, kerning: 1.5, lineLimit: 1, minScale: 0.6 },
    ].slice(0, medium ? 3 : 1).concat(medium ? [] : []),
  };

  const rule = { type: 'rect', fill: RED, frame: { height: 1 } };

  const bigLetter = {
    type: 'text',
    value: first,
    size: medium ? 84 : 64,
    weight: 'black',
    design: 'serif',
    color: RED,
    lineLimit: 1,
    frame: { alignment: 'topLeading' },
    padding: { top: medium ? -14 : -10 },
  };

  const wordBlock = {
    type: 'vstack',
    align: 'leading',
    spacing: 1,
    children: [
      { type: 'text', value: rest, size: medium ? 26 : 20, weight: 'semibold', design: 'serif', italic: true, color: DEEP, lineLimit: 1, minScale: 0.5 },
      { type: 'text', value: String(d.partOfSpeech || '').toLowerCase(), size: 11, design: 'serif', italic: true, color: RED, lineLimit: 1 },
    ],
  };

  const titleRow = {
    type: 'hstack',
    align: 'bottom',
    spacing: 2,
    children: [
      bigLetter,
      { type: 'vstack', align: 'leading', children: [wordBlock], padding: { bottom: medium ? 10 : 6 }, frame: { maxWidth: 'infinity', alignment: 'leading' } },
    ],
    frame: { height: medium ? 62 : 48, alignment: 'leading' },
  };

  const definition = {
    type: 'text',
    value: d.definition,
    size: medium ? 12 : 10.5,
    design: 'serif',
    color: DEEP,
    lineLimit: medium ? 3 : 4,
    minScale: 0.8,
  };

  const example = medium && {
    type: 'hstack',
    spacing: 6,
    align: 'top',
    children: [
      { type: 'rect', fill: RED, frame: { width: 2 } },
      { type: 'text', value: '\u201C' + String(d.example || '') + '\u201D', size: 11, design: 'serif', italic: true, color: RED, lineLimit: 2, minScale: 0.8 },
    ],
    frame: { maxWidth: 'infinity', alignment: 'leading' },
    fixedHeight: undefined,
  };
  if (example) delete example.fixedHeight;

  return {
    type: 'vstack',
    align: 'leading',
    spacing: medium ? 5 : 4,
    padding: { top: 12, bottom: 12, leading: 14, trailing: 14 },
    frame: { maxWidth: 'infinity', maxHeight: 'infinity', alignment: 'topLeading' },
    widgetBackground: { gradient: 'linear', stops: [CREAM, '#EADCBE'], from: 'top', to: 'bottom' },
    children: [
      {
        type: 'hstack',
        children: medium
          ? [
              { type: 'text', value: 'WORD OF THE DAY', size: 8, weight: 'semibold', design: 'serif', color: RED, kerning: 1.5, lineLimit: 1 },
              { type: 'spacer' },
              { type: 'text', value: formatDate(d.date), size: 8, weight: 'semibold', design: 'serif', color: RED, kerning: 1.5, lineLimit: 1 },
            ]
          : [
              { type: 'text', value: formatDate(d.date), size: 8, weight: 'semibold', design: 'serif', color: RED, kerning: 1.5, lineLimit: 1 },
              { type: 'spacer' },
            ],
      },
      rule,
      titleRow,
      definition,
      example,
      { type: 'spacer' },
    ],
  };
}
